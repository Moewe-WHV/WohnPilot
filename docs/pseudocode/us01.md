IF logged_in == TRUE THEN
    DECLARE name : STRING              ### Übernehme die daten aus datenbank
    DECLARE address : STRING
    DECLARE konto : STRING             ### Kontodata für alles was noch kommt!
    DECLARE input : INTEGER

    user_id = SESSION.user_id

    name = DB_GET(user_id, "name")     ### Daten aus Datenbank laden
    address = DB_GET(user_id, "address")
    konto = DB_GET(user_id, "konto")

    ### Kontodaten anzeigen
    OUTPUT name                        ### Ausgabe der daten
    OUTPUT adress
    OUTPUT konto

    OUTPUT "Nächste Optionen:
    1. Edit Name                       ### Menü screen
    2. Edit Address
    3. Edit Kontodata
    4. Abbrechen"

    INPUT input                         ### Die Auswahl des Users

    IF input == 1 THEN
        CALL edit_name(user_id)
    ELSE IF input == 2 THEN
        CALL edit_address(user_id)
    ELSE IF input == 3 THEN
        CALL edit_konto(user_id)
    ELSE IF input == 4 THEN
        CALL user_konto()               ### übergabe in andere Funktionen
    ELSE
        OUTPUT "Ungültige Eingabe."
    ENDIF

ELSE
    OUTPUT "Bitte einloggen, um das Konto zu sehen."  ### Wenn nicht logged_in werden auch 
                                                      ### keine Daten angezeigt
ENDIF


########################################

FUNCTION edit_name(user_id)

    DECLARE new_name : STRING
    DECLARE success : BOOLEAN

    OUTPUT "Neuen Namen eingeben:"
    INPUT new_name

    IF new_name == "" THEN
        OUTPUT "Name darf nicht leer sein."
        RETURN
    ENDIF

    success = DB_UPDATE(user_id, "name", new_name)

    IF success == TRUE THEN
        OUTPUT "Name erfolgreich gespeichert."
    ELSE
        OUTPUT "Fehler beim Speichern."
    ENDIF

END FUNCTION


########################################


FUNCTION edit_address(user_id)

    DECLARE new_address : STRING
    DECLARE success : BOOLEAN

    OUTPUT "Neue Adresse eingeben:"
    INPUT new_address

    IF new_address == "" THEN
        OUTPUT "Adresse darf nicht leer sein."
        RETURN
    ENDIF

    success = DB_UPDATE(user_id, "address", new_address)

    IF success == TRUE THEN
        OUTPUT "Adresse erfolgreich gespeichert."
    ELSE
        OUTPUT "Fehler beim Speichern."
    ENDIF

END FUNCTION

########################################

FUNCTION edit_konto(user_id)

    DECLARE new_konto : STRING
    DECLARE success : BOOLEAN

    OUTPUT "Neue Kontodaten eingeben:"

    IF new_konto == "" THEN

    ENDIF

        OUTPUT "Kontodaten erfolgreich gespeichert."

END FUNCTION
